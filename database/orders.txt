-- Tabele zamówień --
CREATE TABLE IF NOT EXISTS orders (
    id SERIAL PRIMARY KEY,
    total_price NUMERIC(10, 2) NOT NULL,
    created_at TIMESTAMP DEFAULT (NOW() AT TIME ZONE 'Europe/Warsaw'),
    customer_name VARCHAR(100) NOT NULL DEFAULT 'Klient',
    customer_email VARCHAR(100),
    customer_phone VARCHAR(20) NOT NULL DEFAULT '',
    delivery_type VARCHAR(20) NOT NULL DEFAULT 'pickup',
    delivery_street VARCHAR(150),
    delivery_city VARCHAR(100),
    delivery_notes TEXT,
    delivery_fee NUMERIC(10, 2) NOT NULL DEFAULT 0.00,
    scheduled_date DATE NOT NULL DEFAULT CURRENT_DATE,
    scheduled_time TIME NOT NULL DEFAULT CURRENT_TIME,
    status VARCHAR(20) NOT NULL DEFAULT 'pending'
);

CREATE TABLE IF NOT EXISTS order_items (
    id SERIAL PRIMARY KEY,
    order_id INT REFERENCES orders(id) ON DELETE CASCADE,
    dish_id INT NOT NULL,
    quantity INT NOT NULL,
    special_notes TEXT,
    price NUMERIC(10, 2) NOT NULL,
    dish_name VARCHAR(100) NOT NULL
);
