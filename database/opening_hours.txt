-- Godziny otwarcia --
CREATE TABLE IF NOT EXISTS opening_hours (
  id SERIAL PRIMARY KEY,
  days_label VARCHAR(50) NOT NULL,
  open_time TIME NOT NULL,
  close_time TIME NOT NULL,
  display_order INT NOT NULL
);
